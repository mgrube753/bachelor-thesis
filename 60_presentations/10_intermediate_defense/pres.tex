\begin{frame}{Forschungsfragen}
    \begin{itemize}
        \item \textbf{RQ1:} To what extent do \textcolor{RedOrange!80}{LLMs adhere} to the content of diverse provided \textcolor{RedOrange!80}{instructional texts} when generating questions? \vfill
        \item \textbf{RQ2:} How does the \textcolor{RedOrange!80}{relationship} between diverse question formats and Bloom's Taxonomy levels influence the \textcolor{RedOrange!80}{pedagogical effectiveness} of LLM-generated questions?
    \end{itemize}
\end{frame}

\begin{frame}{Bloom's revised Taxonomy}
    \begin{itemize}
        \item Nützliche Grundlage für die Analyse von Fragen
        \item \textbf{Kognitive Hierarchie:} 6-stufiges Framework zur Klassifizierung von Lernzielen \cite{krathwohl_revision_2002}:
        \begin{enumerate}
            {\footnotesize
            \item \textcolor{blue}{Remembering} -- Abruf von Faktenwissen
            \item \textcolor{blue}{Understanding} -- Bedeutung von Anweisungen erkennen
            \item \textcolor{blue}{Applying} -- Wissen in gegebenen Situationen anwenden
            \item \textcolor{blue}{Analyzing} -- Material in Komponenten zerlegen
            \item \textcolor{blue}{Evaluating} -- Urteile basierend auf Kriterien fällen
            \item \textcolor{blue}{Creating} -- Elemente zu Ganzem zusammenfügen
            }
        \end{enumerate}
        \item \textbf{Pädagogische Relevanz:} Ermöglicht systematische Bewertung kognitiver Anforderungen von Fragen
    \end{itemize}
\end{frame}

\begin{frame}{Herausforderungen aus Literatur}
    \begin{itemize}
        \item \textbf{Veraltete Modelle} -- Kein Reasoning
        % pre-thinking basiert auf dem user prompt, damit das modell verstehen kann, was der user will
        \item \textbf{Halluzinationen} -- Falsche Antworten
        % kann man minimieren durch kontext hinzugeben. pre-thinking kann auch dafür sorgen, dass das modell den kontext und die anforderungen hinterfragt
        \item \textbf{Inhaltstreue} -- Forschung nötig
        % es mangelt an systemen, welche die inhaltstreue von LLMs evaluieren
        \item \textbf{Metriken im Fokus} -- Quantitative Evaluation
        % Problematisch, da grundlegend linguistische Metriken
        \item \textbf{Expertenbewertung} -- Qualitative Evaluation 
        % selten performt, weil es aufwendig ist, aber notwendig für geeignete Auswertung
        \item \textbf{Bloom} -- Hohe kognitive Levels
        % dies sind die schwersten Levels, welche LLMs nicht gut beherrschen in related work
        \item \textbf{Prompting} -- Einfluss auf Ergebnisse
        % Auch ich bin kein prompt engineer, jedoch gibt es viele methoden, um LLMs zu steuern
        % kontext geben, rollenzuweisung, beispiellösungen geben --> in-context learning, special marks verwenden für mehr "übersicht" und struktur, auch fine-tuning ist möglich, aber auch aufwendig
    \end{itemize}
\end{frame}

\section{Ergebnisse \& Diskussion}

\begin{frame}{Hypothesen -- RQ1}
    \begin{itemize}
        \item \textbf{H1:} The usage of a \textcolor{RedOrange!80}{complex prompt} with \textcolor{RedOrange!80}{detailed instructions}, in contrast to a common one, will lead to a \textcolor{RedOrange!80}{higher} content fidelity of the generated questions with respect to the source material, and also \textcolor{RedOrange!80}{higher} final scoring for each question. \vfill
        \item \textbf{H1':} \textcolor{RedOrange!80}{Both} the common and complex prompts will lead to a \textcolor{RedOrange!80}{similar} content fidelity of the generated questions with respect to the source material, and also \textcolor{RedOrange!80}{similar} final scoring for each question.
    \end{itemize}
\end{frame}
