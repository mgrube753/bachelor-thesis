\begin{frame}[allowframebreaks]{Motivation \& Problemstellung}
    \begin{itemize}
        \item \textbf{Motivation:} 
        \begin{itemize}
            \item Aktives Lernen $\to$ gezielte Fragen
            \item Anspruchsvolle Fragen $\to$ zeitaufwendig \cite{steuer_i_2021,vu_chatgpt-based_2024} 
        \end{itemize}
        \vspace{.5em}
        \item \textbf{Potenzial von LLMs:} Automatisierte Fragen- generierung $\to$ bessere Lehrmaterialien-Erstellung + personalisiertes Lernen \cite{naveed_comprehensive_2024}
        \framebreak

        \item \textbf{Problemstellung:} Qualität oft unklar $\to$ mangelnde systematische Evaluation über linguistische Tests hinaus \cite{horbach_linguistic_2020} 
        % da auch die pädagogische Eignung bewertet werden muss über experten, was bis dato selten gemacht wurde
        \begin{itemize}
            \item Selten pädagogische Bewertung \cite{zhuge_twinstar_2025,blobstein_angel_2023,scaria_how_2024}
            % manchmal sind wirklich nur linguistische Metriken genutzt worden, oder eben Expertenbewertung ohne Bloom's Taxonomy
            \item Mangelnde Performance bei höheren kognitiven Levels \cite{cheng_treequestion_2024,elkins_how_2023,scaria_how_2024}
            % wurde zumindest festgestellt*, die genutzten Modelle sind auch nicht die neuesten
            % angel gab es z.b.: gpt-3.5 basiert, super relatedness zu context (selbst bewertet), aber sehr niedrige kognitive levels (human eval)
            \item Relevanz $\neq$ Inhaltstreue \cite{elkins_how_2023,hang_mcqgen_2024,mi_comparative_2024,scaria_how_2024}
            % korrektheit von Fragen ist elementar in Bildung
            % Und von interesse ist, wie man gut dies in form von inhaltstreue umsetzen kann
            % inhaltstreue zu gegebenem text wird aber oft nur als future work erwähnt
        \end{itemize}
            % content fidelity an sich immer nur durch experten bewertet, sehr selten

Semantic similarity ist eine Möglichkeit, an sich wurden aber diverse Metriken nur genutzt zwischen generiertem und gold-standard fragen/antworten

\begin{frame}{Herausforderungen der Literatur}
    \begin{itemize}
        \item \textbf{Veraltete Modelle} -- Kein Reasoning
        % pre-thinking basiert auf dem user prompt, damit das modell verstehen kann, was der user will
        % wurde in der literatur noch nicht genutzt, sondern outdated models wie gpt-4/gpt-3.5/llama2
        \item \textbf{Halluzinationen} -- Falsche Antworten
        % kann man minimieren durch kontext hinzugeben. pre-thinking kann auch dafür sorgen, dass das modell den kontext und die anforderungen hinterfragt
        \item \textbf{Inhaltstreue} -- Forschung nötig
        % inhaltstreue ist eher nebensächlich [faraby, doughty], mehr Fokus auf Bloom, linguistische Metriken, Expertenrubriken mit anderem Fokus
        % es mangelt an systemen, welche die inhaltstreue von LLMs evaluieren
        % inhaltstreue ist ein wenig erforschtes feld bei fragengenerierung
        \item \textbf{Metriken im Fokus} -- Quantitative Evaluation
        % Problematisch, da grundlegend linguistische Metriken
        % expertenbewertung ist eine notwendige evaluation bei fragengenerierung
        \item \textbf{Expertenbewertung} -- Qualitative Evaluation 
        % selten ausgiebig performt, weil es aufwendig ist
        \item \textbf{Bloom's Taxonomy} -- Hohe kognitive Levels
        % dies sind die schwersten Levels, welche LLMs nicht gut beherrschen in related work
        % nicht immer wird bei fragen bloom benutzt, und wenn dann sind die modelle eher schlecht im generieren von höheren kognitiven levels, egal ob man informationen für die levels gibt oder nicht
        \item \textbf{Prompting} -- Einfluss auf Ergebnisse
        % Auch ich bin kein prompt engineer, jedoch gibt es viele methoden, um LLMs zu steuern
        % kontext geben, rollenzuweisung, beispiellösungen geben --> in-context learning, special marks verwenden für mehr "übersicht" und struktur, auch fine-tuning ist möglich, aber auch aufwendig
        % gewisse literatur hatte minimale Prompts aber mit Beispieloutputs übergeben, andere haben auch komplexe Prompts verwendet
    \end{itemize}
\end{frame}

\section{Exp.\ 1 Methodik}

\begin{frame}[allowframebreaks]{Experimentelles Design}
    \begin{itemize}
        \item \textbf{Ziel:} Bewertung der Inhaltstreue LLM-\\ generierter Fragen + Qualität
        \item \textbf{Materialien:} 3 Quellen -- ISO-OSI-Modell
        \begin{itemize}
            \item Script -- Prof. Cap's \enquote{Referenzarchitekturen} (DE)
            \item Transcript -- Audio-zu-Text (DE)  
            \item \enquote{Computer Networks} -- Textbook (EN)
        \end{itemize}
        \item \textbf{Exp 1a:} Originaltexte 
        \item \textbf{Exp 1b:} Manipuliertes Script 
        \framebreak
        \item 4 Large Language Models: 
            \begin{itemize}
                \item \textbf{DeepSeek R1:} Jan 2025 -- manuell
                \item \textbf{Claude 3.7 Sonnet:} Feb 2025 -- API
                \item \textbf{Gemini 2.5 Flash:} Mar 2025 -- API
                \item \textbf{OpenAI o3:} Apr 2025 -- API
            \end{itemize}
        \item 2 Prompt-Strategien:
            \begin{itemize}
                \item \textbf{Einfach:} Weniger Anweisungen, keine Strukturierung, aber Formatvorlage + Material
                % generiere zum iso osi modell
                % korrekte Antwort klar gekennzeichnet (eine oder eben aus möglichkeiten)
                % kein abgrenzen von paragraphen (strukturierung)
                % klar und präzise Frage, kein weiterer Text soll generiert werden
                \item \textbf{Komplex:} Rollenzuweisung, detaillierte Anweisungen, fordert nicht-triviales und kritisches Denken, Struktur durch Docstrings, Format-Vorlage + Material 
                % format für einheitlichen Output bei beiden Prompts --> war das Problem beim einfachen Prompt --> Rerun nötig gewesen
                % + source material
                % was war drin:
                % erfahrener experte im generieren von kognitiv anspruchsvollen fragen, welche nicht triviales und analytisches Denken fordern
                % untersuchen des iso osi texts, erfassen des wichtigen
                % soll nur fliesstext statt markdown generieren, nichts anderes soll generiert werden
            \end{itemize}
        % diese haben keinen festen fragetypen vorgegeben. 
        % dies ermöglicht, ergebnisse ohne die eine zusätzliche Variable zu verfälschen, da fokus auf inhaltstreue liegt
        % kann eine fundierung für verbessern der prompts in experiment 2 sein, wo teils auch fragetyp nicht angegeben wird, wie im komplexen prompt auch
        % dafür ist halt RQ2 da, um mit sowas umzugehen
        % wir haben eine geeignete vergleichbarkeit durch die lang durchdachte auswertungsrubrik. wäre diese oberflächlich, wäre die vergleichbarkeit wesentlich geringer
        % auch die quantitative umsetzung ist recht formatunabhängig (habe die falschen antwortmöglichkeiten abgefiltert), kann ja mal zählen wie viele fragen von typ x generiert wurden --> möglicher erkenntnisgewinn bzgl. präferenzen bei fragestellung (multiple-choice bei exp1)
        % dann kann man in exp2 gucken, was auf den jeweiligen kognitiven leveln passiert, wenn man dort fragetyp vs. bloom level angibt, wie gut das funktioniert
    \end{itemize}
\end{frame}

\begin{frame}[allowframebreaks]{Evaluations-Methodik}
    \begin{itemize}
        \item \textbf{Quantitativ:}
        \begin{itemize}
            \item \textbf{Semantic Similarity:} Generierte Frage vs. Originaltext
            \begin{itemize}
                \item[$\to$] Embedding-Modell \& Cosine Similarity \vspace{-.6em}
                % cosine sim
                % Bsp: (3,2) und (4,1) --> 3*4 + 2*1 / (sqrt(3^2+2^2) * sqrt(4^2+1^2)) = 14 / (sqrt(13) * sqrt(17)) = 0.94, sonst a mal b / ||a|| * ||b||
            \end{itemize}
            \begin{align*}
                \hspace{-3em}\cos(\vec{a}, \vec{b}) &= \frac{\vec{a} \cdot \vec{b}}{||\vec{a}|| \cdot ||\vec{b}||} 
            \end{align*}
            % zu beginn der transformer-ära gab es encoder-decoder transformer, wo encoder den input in eine mathematische Repräsentation mit Bedeutung umwandelt
            % dann generiert der decoder basierend auf diesem embedding den output token-weise
            % hat sich aber eher gesplittet in encoder-only und decoder-only transformer
            % decoder-only: gpt, llama, gemini: können text generieren, weniger aufwendiges Training als enc-dec-transformer
            % encoder-only: bert, roberta: können nur text klassifizieren, nicht generieren --> sentence transformers kamen dann auch --> diese Embeddings sind vegleichbar
            \item \textbf{Adherence Score:} o3 \& Claude 3.7 Sonnet Prompting
            % temperatur auf 0 für weniger randomness in Antwort, geht aber nicht bei thinking mode...
            % bekommt Rolle eines Experten im Bewerten von metrischen Fragen, Frage studieren, iso osi kontext text studieren
            % kurze Bewertungsrubrik 0 bis 1, Ausgabe einer Dezimalzahl mit 2 Nachkommastellen, ohne Kommentare etc.
        \end{itemize}
        % \item quantitative mit metriken (semantic sim mit Embedding Model, adherence via llm)
        \item \textbf{Qualitativ:} Prof. Cap + 4 Experten des IuK 
        \begin{itemize}
            \item \textbf{Fleiss Kappa:} Inter-Rater Reliability
            \item \textbf{Bewertungsrubrik:} 0-10 Punkte\,/\,Kategorie \& Frage:
            \item[] \textit{\textcolor{gray}{Relevanz, Klarheit, Beantwortbarkeit, Herausfordernd \& Korrektheit vs. Umgang mit Manipulation}}
        \end{itemize}
    \end{itemize}
\end{frame}

\section{Fazit \& Ausblick}

\begin{frame}{Hauptergebnisse zu RQ1 -- Exp. 1a}
    \begin{itemize}
        \item Nicht vollständig, da Expertenbewertung aussteht
        \item Jedoch moderate Ergebnisse
        \begin{itemize}
            \item DeepSeek R1 am schwächsten bei cossim, google/anthropic gut, openai okay
            \item Adherence: llms nah beieinander stattdessen
            \item Script > Tanenbaum > Transcript bei cossim
            \item Bringt somit bei Layer-Analyse auch viel Streuung bei 1-4, in adherence nicht spürbar
            % transcript 1-4 ziehen die Werte runter, da die Quellen sehr lang sind
            % Script schneidet durch die sehr kurzen Formulierungen besser ab
            \item unlogisch hingegen adherence score: alle sources ähnlich
            \item Komplexer Prompt > Einfacher Prompt, aber nicht signifikant
            \item unlogisch hingegen adherence: andersherum, relative deutlich
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Hauptergebnisse zu RQ1 -- Exp. 1b}
    \begin{itemize}
        \item Nicht vollständig, 1b ist auch eher ein sekundärer Case der Forschungsfrage
        \item Auch hier, DeepSeek R1 am schwächsten, aber bei beiden Metriken (und streut bei adherence sehr)
        \item Adherence bei diesem source material wieder viel besser nach oben hin
        \item Einfacher Prompt > Komplexer Prompt, bei adherence viel diversität in den Werten
        \item cossim bei layers geringer grundsätzlich ggü. 1a, adherence bei 1 und 3 mehr nach unten
        \item Cossim bleibt stets hoch, aber nicht erkennbar, ob LLMs dem Text stur folgen, oder die Manipulation merken
        \item Adherence Score fällt aber pro LLM nach unten
        \item Dadurch, dass die Werte sich nicht extrem von denen aus Exp. 1a unterscheiden, ist an sich das Folgen des Textes gegeben
        \item Ergebnisse jedoch schwer nachvollziehbar, da Phrasen vorkommen wie \enquote{Laut dem widersprüchlichen Text, was sind ...}
        \item Expertenanalyse von Bedeutung hier
    \end{itemize}
\end{frame}
