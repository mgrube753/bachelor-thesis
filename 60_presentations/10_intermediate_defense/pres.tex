        % es mangelt an systemen, welche die inhaltstreue von LLMs evaluieren
        % inhaltstreue ist ein wenig erforschtes feld bei fragengenerierung
        \item \textbf{Metriken im Fokus} -- Quantitative Evaluation
        % Problematisch, da grundlegend linguistische Metriken
        % expertenbewertung ist eine notwendige evaluation bei fragengenerierung
        \item \textbf{Expertenbewertung} -- Qualitative Evaluation 
        % selten ausgiebig performt, weil es aufwendig ist
        \item \textbf{Bloom's Taxonomy} -- Hohe kognitive Levels
        % dies sind die schwersten Levels, welche LLMs nicht gut beherrschen in related work
        % nicht immer wird bei fragen bloom benutzt, und wenn dann sind die modelle eher schlecht im generieren von höheren kognitiven levels, egal ob man informationen für die levels gibt oder nicht
        \item \textbf{Prompting} -- Einfluss auf Ergebnisse
        % Auch ich bin kein prompt engineer, jedoch gibt es viele methoden, um LLMs zu steuern
        % kontext geben, rollenzuweisung, beispiellösungen geben --> in-context learning, special marks verwenden für mehr "übersicht" und struktur, auch fine-tuning ist möglich, aber auch aufwendig
        % gewisse literatur hatte minimale Prompts aber mit Beispieloutputs übergeben, andere haben auch komplexe Prompts verwendet
    \end{itemize}

        \item 4 Large Language Models: 
            \begin{itemize}
                \item \textbf{DeepSeek R1:} Jan 2025 -- manuell
                \item \textbf{Claude 3.7 Sonnet:} Feb 2025 -- API
                \item \textbf{Gemini 2.5 Flash:} Mar 2025 -- API
                \item \textbf{OpenAI o3:} Apr 2025 -- API
            \end{itemize}
        \item 2 Prompt-Strategien:
            \begin{itemize}
                \item \textbf{Einfach:} Weniger Anweisungen, keine Strukturierung, aber Formatvorlage + Material
                % generiere zum iso osi modell
                % korrekte Antwort klar gekennzeichnet (eine oder eben aus möglichkeiten)
                % kein abgrenzen von paragraphen (strukturierung)
                % klar und präzise Frage, kein weiterer Text soll generiert werden
                \item \textbf{Komplex:} Rollenzuweisung, detaillierte Anweisungen, fordert nicht-triviales und kritisches Denken, Struktur durch Docstrings, Formatvorlage + Material 
                % format für einheitlichen Output bei beiden Prompts --> war das Problem beim einfachen Prompt --> Rerun nötig gewesen
                % + source material
                % was war drin:
                % erfahrener experte im generieren von kognitiv anspruchsvollen fragen, welche nicht triviales und analytisches Denken fordern
                % untersuchen des iso osi texts, erfassen des wichtigen
                % soll nur fliesstext statt markdown generieren, nichts anderes soll generiert werden
            \end{itemize}
        % diese haben keinen festen fragetypen vorgegeben.

        % kann eine fundierung für verbessern der prompts in experiment 2 sein, wo teils auch fragetyp nicht angegeben wird, wie im komplexen prompt auch

\begin{frame}[allowframebreaks]{Evaluations-Methodik}
    \begin{itemize}
        \item \textbf{Quantitativ:}
        \begin{itemize}
            \item \textbf{Semantic Similarity:} Generierte Frage vs. Originaltext
            \begin{itemize}
                \item[$\to$] Embedding-Modell \& Cosine Similarity \vspace{-.6em}
                % cosine sim
                % Bsp: (3,2) und (4,1) --> 3*4 + 2*1 / (sqrt(3^2+2^2) * sqrt(4^2+1^2)) = 14 / (sqrt(13) * sqrt(17)) = 0.94, sonst a mal b / ||a|| * ||b||
            \end{itemize}
            \begin{align*}
                \hspace{-3em}\cos(\vec{a}, \vec{b}) &= \frac{\vec{a} \cdot \vec{b}}{||\vec{a}|| \cdot ||\vec{b}||} 
            \end{align*}
            % zu beginn der transformer-ära gab es encoder-decoder transformer, wo encoder den input in eine mathematische Repräsentation mit Bedeutung umwandelt
            % dann generiert der decoder basierend auf diesem embedding den output token-weise
            % hat sich aber eher gesplittet in encoder-only und decoder-only transformer
            % decoder-only: gpt, llama, gemini: können text generieren, weniger aufwendiges Training als enc-dec-transformer
            % encoder-only: bert, roberta: können nur text klassifizieren, nicht generieren --> sentence transformers kamen dann auch --> diese Embeddings sind vegleichbar
            \item \textbf{Adherence Score:} o3 \& Claude 3.7 Sonnet Prompting
            % temperatur auf 0 für weniger randomness in Antwort, geht aber nicht bei thinking mode...
            % bekommt Rolle eines Experten im Bewerten von metrischen Fragen, Frage studieren, iso osi kontext text studieren
            % kurze Bewertungsrubrik 0 bis 1, Ausgabe einer Dezimalzahl mit 2 Nachkommastellen, ohne Kommentare etc.
        \end{itemize}
        % \item quantitative mit metriken (semantic sim mit Embedding Model, adherence via llm)
        \item \textbf{Qualitativ:} Prof. Cap + 4 Experten des IuK 
        \begin{itemize}
            \item \textbf{Fleiss Kappa:} Inter-Rater Reliability
            \item \textbf{Bewertungsrubrik:} 0-10 Punkte\,/\,Kategorie \& Frage:
            \item[] \textit{\textcolor{gray}{Relevanz, Klarheit, Beantwortbarkeit, Herausfordernd \& Korrektheit vs. Umgang mit Manipulation}}
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{How to: Fleiss Kappa}
    \begin{itemize}
        \item \textbf{Ziel:} Inter-Rater Reliability der Experten durch:
        \begin{align*}
            \kappa &= \frac{p_o - p_e}{1 - p_e}, \text{ mit}
        \end{align*}
        $p_0$ beobachtete Übereinstimmung, \\ $p_e$ erwartete Übereinstimmung durch Zufall, \\
        $N$ Anzahl der Rater, \\
        $n$ Anzahl der Kategorien
        \begin{align*}
            % N anzahl der Rater, n Anzahl der Kategorien, n_{ij} Anzahl der Rater, die Kategorie i für Frage j gewählt haben
            p_o &= \frac{\sum_{i=1}^{N}\sum_{j=1}^{k} n_{ij}^2-N\cdot n}{N\cdot n\cdot (n-1)} \\
            % 0^2 + 3^2 \\ + 1^2 + 2^2 ...
            p_e &= \sum_{i=1}^{n} p_i^2
            % wie viele rater haben in der kategorie i eine zustimmung gegeben (2 kategorien, in der einen 8/21 eintraege, bei der anderen 13/21 einträge)
        \end{align*}
    \end{itemize}
\end{frame}

        \item[$\to$] Expertenbewertung von Bedeutung
    \end{itemize}
\end{frame}

\begin{frame}{Hauptergebnisse zu RQ1 -- Exp. 1a}
    \begin{itemize}
        % \item Nicht vollständig, da Expertenbewertung aussteht
        \item \textbf{Bezüglich der LLMs:}
        \begin{itemize}
            \item[$\to$] \textbf{CosSim:} DeepSeek R1 schwächstes, Gemini bestes
            \item[$\to$] \textbf{Adherence:} Ergebnisse nah beieinander
            % dort sind präferenzen/probleme nicht wirklich erkennbar
        \end{itemize}
        \vspace{.5em}
        \item \textbf{Eingabequellen:}
        \begin{itemize}
            \item[$\to$] \textbf{CosSim:} Script > Tanenbaum > Transcript
            % Script schneidet durch die sehr kurzen Formulierungen besser ab
            % Transcript Streuung bei Layern 1-4 (sehr lang), in adherence nicht spürbar
            \item[$\to$] \textbf{Adherence:} Quellen-übergreifend ähnlich
        \end{itemize}
        \vspace{.5em}
        \item \textbf{Prompt-Strategien:}
        \begin{itemize}
            \item[$\to$] \textbf{CosSim:} Komplexer Prompt > Einfacher Prompt
            % aber nicht signifikant
            \item[$\to$] \textbf{Adherence:} Eher andersherum
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Hauptergebnisse zu RQ1 -- Exp. 1b}
    \begin{itemize}
        % \item Nicht vollständig, 1b ist auch eher ein sekundärer Case der Forschungsfrage
        \item \textbf{Manipulierte Inhalte -- Performance:}
        \begin{itemize}
            \item[$\to$] \textbf{DeepSeek R1:} schwächstes bei beiden Metriken
            % (und streut bei adherence sehr)
            \item[$\to$] \textbf{CosSim:} Stets hoch, Manipulation erkannt?
            % aber nicht erkennbar, ob LLMs dem Text stur folgen, oder die Manipulation merken
            % sobald man über die manipulation hinaus springt mit viel text, ist das scoring wieder niedrig
            % widersprüche in dem semantischen raum sind wahrscheinlich hier das problem
            \item[$\to$] \textbf{Adherence:} Instabil, DeepSeek und OpenAI besonders betroffen im komplexen Prompt
            % alles schwer nachvollziehbar
            % bei adherence viel diversität in den Werten
            \item[$\to$] \textbf{Beide Metriken:} Einfacher Prompt > Komplexer Prompt
        \end{itemize}

        \item \textbf{Kritisch:} \enquote{Laut dem zum ISO-OSI-Modell wider\-sprüchlichen Text, was sind [...]?}
        \item[$\to$] Expertenanalyse von Bedeutung
    \end{itemize}
\end{frame}
