\section{Einführung \& Problem}

\begin{frame}[allowframebreaks]{Motivation \& Problemstellung}
    \begin{itemize}
        \item \textbf{Motivation:} Aktives Lernen $\to$ gezielte Fragen, aber \\ anspruchsvolle Fragen $\to$ zeitaufwendig \cite{steuer_i_2021,vu_chatgpt-based_2024} 
        \vspace{.5em}
        \item \textbf{Potenzial von LLMs:} Automatisierte Fragen- generierung $\to$ Lehrmaterialien verbessern + personalisiertes Lernen \cite{naveed_comprehensive_2024}
        % biancini: used compliance,
        % al faraby: future work shall be avoiding hallucinations by utilizing a framework which checks content fidelity
        % angel: super relatedness to context, but very low bloom levels/adherence
        % doughty: adherence between question and learning content was not assessed 
        % elkins: relevance assessed, more interesting the bloom adherence
        % mcqgen: relevance 100% on experts and llm-based eval (Rag)
        % lamsiyah: mangel an menschlicher bewertung, um e.g. unterschiede zwischen menschlichen und llm-generierten fragen zu evaluieren
        % maity: relevance used in eval
        % mi and li (my rubric): relevance as well!
        % content fidelity an sich immer nur durch experten bewertet, diverse insights (mal 5/5 nur, andere studie ist e.g. divers dabei)
        % scaria how good: nutzt horbach rubrik und relevance ergibt sich aus mehreren kategorie-ergebnissen
        % vu: sagen, dass man content als wissensbasis beilegen soll, um qualitaet zu verbessern
        % yang: halluzination in bildung inakzeptabel, somit RAG wieder
        % twinstar: besides bloom adherence also knowledge relevance, measured automatically but unknown
        \framebreak
        \item \textbf{Problemstellung:} Qualität oft unklar $\to$ mangelnde systematische Evaluation über linguistische Tests hinaus \cite{horbach_linguistic_2020} 
        % da auch die pädagogische Eignung bewertet werden muss
        \vspace{.5em}
        \item \textbf{Forschungslücke:} Bedarf an Untersuchung der Inhaltstreue und Einhalten kognitiver Anforderungsniveaus
    \end{itemize}
\end{frame}

\begin{frame}{Forschungsfragen}
    \begin{itemize}
        \item \textbf{RQ1:} To what extent do \textcolor{RedOrange!80}{LLMs adhere} to the content of diverse provided \textcolor{RedOrange!80}{instructional texts} when generating questions? \vspace{.5em}
        \item \textbf{RQ2:} How does the \textcolor{RedOrange!80}{relationship} between diverse question formats and Bloom's Taxonomy levels influence the \textcolor{RedOrange!80}{pedagogical effectiveness} of LLM-generated questions?
    \end{itemize}
\end{frame}

\begin{frame}{Bloom's revised Taxonomy}
    \begin{itemize}
        \item Nützliche Grundlage für die Analyse von Fragen
        \item \textbf{Kognitive Hierarchie:} Lernzielklassifizierung \cite{krathwohl_revision_2002}:
        \begin{enumerate}
            {\footnotesize
            \item \textcolor{blue}{Remembering} -- Abruf von Faktenwissen
            \item \textcolor{blue}{Understanding} -- Bedeutung von Anweisungen erkennen
            \item \textcolor{blue}{Applying} -- Wissen in gegebenen Situationen anwenden
            \item \textcolor{blue}{Analyzing} -- Material in Komponenten zerlegen
            \item \textcolor{blue}{Evaluating} -- Urteile basierend auf Kriterien fällen
            \item \textcolor{blue}{Creating} -- Elemente zu Ganzem zusammenfügen
            }
        \end{enumerate}
        \item \textbf{Pädagogische Relevanz:} Ermöglicht systematische Bewertung kognitiver Anforderungen von Fragen
    \end{itemize}
\end{frame}

\begin{frame}{Herausforderungen aus Literatur}
    \begin{itemize}
        \item \textbf{Veraltete Modelle} -- Kein Reasoning
        % pre-thinking basiert auf dem user prompt, damit das modell verstehen kann, was der user will
        \item \textbf{Halluzinationen} -- Falsche Antworten
        % kann man minimieren durch kontext hinzugeben. pre-thinking kann auch dafür sorgen, dass das modell den kontext und die anforderungen hinterfragt
        \item \textbf{Inhaltstreue} -- Forschung nötig
        % es mangelt an systemen, welche die inhaltstreue von LLMs evaluieren
        \item \textbf{Metriken im Fokus} -- Quantitative Evaluation
        % Problematisch, da grundlegend linguistische Metriken
        \item \textbf{Expertenbewertung} -- Qualitative Evaluation 
        % selten performt, weil es aufwendig ist, aber notwendig für geeignete Auswertung
        \item \textbf{Bloom's Taxonomy} -- Hohe kognitive Levels
        % dies sind die schwersten Levels, welche LLMs nicht gut beherrschen in related work
        \item \textbf{Prompting} -- Einfluss auf Ergebnisse
        % Auch ich bin kein prompt engineer, jedoch gibt es viele methoden, um LLMs zu steuern
        % kontext geben, rollenzuweisung, beispiellösungen geben --> in-context learning, special marks verwenden für mehr "übersicht" und struktur, auch fine-tuning ist möglich, aber auch aufwendig
        % gewisse literatur hatte minimale Prompts aber mit Beispieloutputs übergeben, andere haben auch komplexe Prompts verwendet
    \end{itemize}
\end{frame}

\section{Exp.\ 1 Methodik}

\begin{frame}[allowframebreaks]{Experimentelles Design}
    \begin{itemize}
        \item \textbf{Ziel:} Bewertung der Inhaltsadhärenz von LLM-\\ generierten Fragen
        \item \textbf{Materialien:} 3 Quellen -- ISO-OSI-Modell
        \begin{itemize}
            \item Script -- Prof. Cap's \enquote{Referenzarchitekturen} (DE)
            \item Transcript -- Audio-zu-Text (DE)  
            \item \enquote{Computer Networks} -- Textbook (EN)
        \end{itemize}
        \item \textbf{Exp 1a:} Nutzt Originaltexte 
        \item \textbf{Exp 1b:} Manipuliertes Script 
        \framebreak
        \item 4 Large Language Models: 
            \begin{itemize}
                \item \textbf{DeepSeek R1:} Jan 2025 -- manuell
                \item \textbf{Claude 3.7 Sonnet:} Feb 2025 -- API
                \item \textbf{Gemini 2.5 Flash:} Mar 2025 -- API
                \item \textbf{OpenAI o3:} Apr 2025 -- API
                % 1a: 4llms, 3inputs, 7layers, 2prompts = 168 fragen
                % 1b: 4llms, 1input, 7layers, 2prompts = 56 fragen
            \end{itemize}
        \item 2 Prompt-Strategien:
            \begin{itemize}
                \item \textbf{Einfach:} Minimale Instruktionen, keine Strukturierung, aber Formatvorlage + Material
                % generiere zum iso osi modell
                % korrekte Antwort klar gekennzeichnet (eine oder eben aus möglichkeiten)
                % klar und präzise Frage, kein weiterer Text soll generiert werden
                \item \textbf{Komplex:} Rollenzuweisung, detaillierte Anweisungen, fordert nicht-triviales und kritisches Denken, Struktur durch Docstrings, Format-Vorlage + Material 
                % format für einheitlichen Output bei beiden Prompts --> war das Problem beim einfachen Prompt --> Rerun nötig gewesen
                % + source material
                % was war drin:
                % erfahrener experte im generieren von kognitiv anspruchsvollen fragen, welche nicht triviales und analytisches Denken fordern
                % untersuchen des iso osi texts, erfassen des wichtigen
                % soll nur fliesstext statt markdown generieren, nichts anderes soll generiert werden
            \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Datenerhebung \& Durchführung}
    \begin{itemize}
        % \item Python, embedding modell, apis ausser deepseek, automatisierte anfragen mit diversem prompting, diversen source materials, ...
        \item \textbf{Python}: automatisierte API-Ansteuerungen
        \item \textbf{Manuell}: DeepSeek R1
        \item \textbf{Material}: Layer-weise Aufteilung
        \item \textbf{Prompt-Variation}: jeweils 1 Frage
        \item \textbf{Resultate}: 
            \begin{itemize}
                \item \textbf{Exp 1a:} 168 Fragen
                \item \textbf{Exp 1b:} 56 Fragen

    \end{itemize}

    \end{align*}
\end{frame}

\begin{frame}{Evaluations-Methodik}
    \begin{itemize}
        \item \textbf{Quantitativ:}
        \begin{itemize}
            \item \textbf{Semantic Similarity:} Vergleich generierte Frage vs. Originaltext $\to$ Embedding-Modell \& Cosine Similarity
            % zu beginn der transformer-ära gab es encoder-decoder transformer, welche den input in eine mathematische Repräsentation mit Bedeutung umwandeln
            % der decoder generiert dann basierend auf diesem embedding den output token-weise
            % hat sich aber eher gesplittet in encoder-only und decoder-only transformer
            % decoder-only: gpt, llama, gemini: können text generieren, weniger aufwendiges Training als enc-dec-transformer
            % encoder-only: bert, roberta: können nur text klassifizieren, nicht generieren --> sentence transformers kamen dann auch --> diese Embeddings sind vegleichbar
            \item \textbf{Adherence:} Inhaltstreue bewerten durch 2x Prompting\,/\,Frage an Gemini 2.5 Flash
            % temperatur auf 0 für weniger randomness in Antwort
            % bekommt Rolle eines Experten im Bewerten von metrischen Fragen, Frage studieren, iso osi kontext text studieren
            % kurze Bewertungsrubrik 0 bis 1, Ausgabe einer Dezimalzahl mit 2 Nachkommastellen, ohne Kommentare etc.
        \end{itemize}
        % \item quantitative mit metriken (semantic sim mit Embedding Model, adherence via llm)
        \item \textbf{Qualitativ:} Prof. Cap + 4 Experten des IuK 
        \begin{itemize}
            \item Bewertungsrubrik: 0-10 Punkte\,/\,Frage
            \item Relevanz, Klarheit, Beantwortbarkeit, Herausforderung, Korrektheit
        \end{itemize}
    \end{itemize}
\end{frame}

\section{Ergebnisse \& Diskussion}

\begin{frame}{Hypothesen -- RQ1}
    \begin{itemize}
        \item \textbf{H1:} The usage of a \textcolor{RedOrange!80}{complex prompt} with \textcolor{RedOrange!80}{detailed instructions}, in contrast to a common one, will lead to a \textcolor{RedOrange!80}{higher} content fidelity of the generated questions with respect to the source material, and also \textcolor{RedOrange!80}{higher} final scoring for each question. \vfill
        \item \textbf{H1':} \textcolor{RedOrange!80}{Both} the common and complex prompts will lead to a \textcolor{RedOrange!80}{similar} content fidelity of the generated questions with respect to the source material, and also \textcolor{RedOrange!80}{similar} final scoring for each question.
    \end{itemize}
\end{frame}
